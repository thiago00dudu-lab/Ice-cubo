const axios = require("axios"); const ACCESS_TOKEN = process.env.ACCESS_TOKEN; module.exports = async (req, res) => { // ========================== // CRIAR PIX // ========================== if (req.method === "POST" && req.url === "/api/pix") { const { valor, email } = req.body; try { const response = await axios.post( "https://api.mercadopago.com/v1/payments", { transaction_amount: Number(valor), description: "Depósito Ice Blue", payment_method_id: "pix", payer: { email: email } }, { headers: { Authorization: `Bearer ${ACCESS_TOKEN}` } } ); return res.status(200).json({ id: response.data.id, qr_code: response.data.point_of_interaction.transaction_data.qr_code, qr_code_base64: response.data.point_of_interaction.transaction_data.qr_code_base64 }); } catch (error) { return res.status(500).json({ erro: "Erro ao gerar PIX" }); } } // ========================== // WEBHOOK // ========================== if (req.method === "POST" && req.url === "/api/webhook") { console.log("Webhook recebido:", req.body); return res.status(200).send("ok"); } // ========================== // SAQUE (simulado) // ========================== if (req.method === "POST" && req.url === "/api/saque") { const { valor } = req.body; return res.status(200).json({ status: "Saque solicitado", valor: valor }); } return res.status(404).json({ erro: "Rota não encontrada" }); };
