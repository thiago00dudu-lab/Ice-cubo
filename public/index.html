let token = ""; async function fazerLogin() { const usuario = document.getElementById("usuario").value; const senha = document.getElementById("senha").value; const res = await fetch("/api/login", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ username: usuario, password: senha }) }); if (res.ok) { const dados = await res.json(); token = dados.token; document.getElementById("login-card").classList.add("hidden"); document.getElementById("painel").classList.remove("hidden"); document.getElementById("nome-exibicao").innerText = usuario; document.getElementById("saldo").innerText = dados.blue; } else { alert("Erro ao entrar. Verifique os dados."); } } async function comprar() { const valor = document.getElementById("valor").value; if (!valor) return alert("Digite um valor!"); const res = await fetch("/api/buy", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": token }, body: JSON.stringify({ amount: valor }) }); if (res.ok) { const dados = await res.json(); // Abre o link do PIX (invoiceUrl) em uma nova aba if (dados.invoiceUrl) { window.open(dados.invoiceUrl, "_blank"); } else { alert("PIX gerado! Verifique seu app do Asaas."); } } else { alert("Erro ao gerar pagamento."); } }
